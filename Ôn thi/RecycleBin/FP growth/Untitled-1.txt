	/**
	 * Mine an FP-Tree having more than one path.
	 * @param tree  the FP-tree
	 * @param prefix  the current prefix, named "alpha"
	 * @param mapSupport the frequency of items in the FP-Tree
	 * @throws IOException  exception if error writing the output file
	 */
	private void fpclose(FPTree tree, int [] prefix, int prefixLength, int prefixSupport, Map<Integer, Integer> mapSupport) throws IOException {
        
        //======= DEBUG ========
		if(DEBUG) {
			System.out.print("###### Prefix: ");
			for(int k=0; k< prefixLength; k++) {
				System.out.print(prefix[k] + "  ");
			}
			System.out.println("\n");
			System.out.println(tree);
		}	
        //========== END DEBUG =======


        // Đầu tiên, chúng ta kiểm tra xem FPtree có chứa một đường dẫn duy nhất hay không
        // Chúng ta sẽ kiểm tra xem FPtree có chứa một đường dẫn duy nhất hay không
        boolean singlePath = true;

        // Biến này được sử dụng để đếm số lượng mục trong đường dẫn duy nhất (nếu có)
        // một + độ dài tiền tố
		int position = prefixLength;

		if(tree.root.childs.size() > 1) {
            // Nếu gốc có nhiều hơn một nhánh con, thì đó không phải là một đường dẫn duy nhất
			singlePath = false;
		}
        else {
            // Nếu nút gốc chỉ có đúng một nút con, chúng ta cần kiểm tra đệ quy các nút con của nút con đó để xem chúng cũng có một nút con hay không.
            // của nút con đó để xem chúng có nút con nào khác không.
			FPNode currentNode = tree.root.childs.get(0);
			while(true){

				// Nếu phần tử con hiện tại có nhiều hơn một phần tử con, thì đó không phải là một đường dẫn duy nhất!
				if(currentNode.childs.size() > 1) {
					singlePath = false;
					break;
				}

				// Ngược lại, chúng ta sao chép mục hiện tại trong bộ đệm và chuyển nó đến phần tử con.
				itemsetBuffer[position] = currentNode.itemID;
				countBuffer[position] = currentNode.counter;
				position++;

				// Nếu nút này không có nút con, điều đó có nghĩa là đây là điểm cuối của đường dẫn này
                // và đây là một đường dẫn đơn, vì vậy chúng ta dừng lại
				if(currentNode.childs.size() == 0) {
					break;
				}

				currentNode = currentNode.childs.get(0);
			}
		}
		
        // Trường hợp 1: Cây FPtree chứa một đường dẫn duy nhất
        // Nếu đường dẫn này có đủ hỗ trợ:
		if(singlePath && countBuffer[position-1] >= minSupportRelative){
			
            // Tạo tất cả các CFI từ đường dẫn này
            // Đối với mỗi CFI X được tạo ra, chúng ta sẽ kiểm tra xem X đã được đóng hay chưa
            // bằng cách xem cây CFI. Nếu có, chúng ta sẽ chèn X vào
            // cây CFI
			for(int i = prefixLength; i <= position; i++) {

				// nếu là phần tử cuối
				if(i == position) {
					int pathSupport = countBuffer[i-1];
					
					// nếu tập hợp mục hiện tại vượt qua kiểm tra đóng
                    // chúng ta lưu tập hợp mục này dưới dạng tập hợp mục đã đóng
					int[] headWithP = new int[i];
					System.arraycopy(itemsetBuffer, 0, headWithP, 0, i);
					sortOriginalOrder(headWithP, i);
					
					if(cfiTree.passSubsetChecking(headWithP, i, pathSupport)) {
						saveItemset(headWithP, i, pathSupport);
					}
				}
                else {

					// nếu bộ đếm của phần tử ở vị trí thứ i+1 khác với bộ đếm của phần tử ở vị trí thứ i:
					if(i > 0 && countBuffer[i-1] !=0 && countBuffer[i-1] != countBuffer[i]) {
					
                    	int pathSupport = countBuffer[i-1];  // NEW
						
						// nếu tập hợp mục hiện tại vượt qua kiểm tra đóng
                        // chúng ta lưu tập hợp mục này dưới dạng tập hợp mục đã đóng
						int[] headWithP = new int[i];
						System.arraycopy(itemsetBuffer, 0, headWithP, 0, i);
						sortOriginalOrder(headWithP, i);
						
						if(cfiTree.passSubsetChecking(headWithP, i, pathSupport)) {
                            // Nếu tập hợp các mục kết thúc ở vị trí thứ i vượt qua việc kiểm tra tính đóng,
                            // chúng ta lưu tập hợp các mục kết thúc ở vị trí thứ i dưới dạng một tập hợp các mục đã đóng
							saveItemset(headWithP, i, pathSupport);
						}
					}
				}
				
			}
		}
        else {
			
            // Trường hợp 2: Có nhiều đường dẫn.
            // Đối với mỗi mục thường xuyên xuất hiện trong danh sách bảng tiêu đề của cây theo thứ tự ngược lại. (theo thứ tự giảm dần về độ hỗ trợ...)
			for(int i = tree.headerList.size()-1; i>=0; i--){
				// lấy item
				Integer item = tree.headerList.get(i);
				
				// Lấy số hỗ trợ của item
				int support = mapSupport.get(item);
				
				// Tính toán mức độ hỗ trợ của tiền tố mới beta
				int betaSupport = (prefixSupport < support) ? prefixSupport: support;

                // Tạo số Beta bằng cách nối mục với tiền tố Alpha hiện tại
				prefix[prefixLength] = item;
				countBuffer[prefixLength] = betaSupport;
				
                // === (A) Xây dựng cơ sở mẫu điều kiện của beta ===
                // Đó là một cơ sở dữ liệu con bao gồm tập hợp các đường dẫn tiền tố
                // trong cây FP cùng xuất hiện với mẫu tiền tố.
				List<List<FPNode>> prefixPaths = new ArrayList<List<FPNode>>();
				FPNode path = tree.mapItemNodes.get(item);
				
                // Bản đồ để đếm mức độ hỗ trợ của các mục trong cây tiền tố có điều kiện
                // Khóa: mục Giá trị: hỗ trợ
				Map<Integer, Integer> mapSupportBeta = new HashMap<Integer, Integer>();
				
				while(path != null){
					// nếu đường dẫn không chỉ là nút gốc
					if(path.parent.itemID != -1){

                        // Tạo prefixPath
                        List<FPNode> prefixPath = new ArrayList<FPNode>();

                        // Thêm nút này.
                        prefixPath.add(path);

                        // LƯU Ý: Chúng ta thêm nó chỉ để duy trì sự hỗ trợ của nó,
                        // thực tế nó không nên là một phần của prefixPath

						// ####
						int pathCount = path.counter;
						
						// Thêm đệ quy tất cả các nút cha của nút này.
						FPNode parent = path.parent;
						while(parent.itemID != -1){
							prefixPath.add(parent);
							
                            // VỚI MỖI MẪU, CHÚNG TÔI CŨNG CẬP NHẬT HỖ TRỢ MẶT HÀNG CÙNG LÚC
                            // nếu lần đầu tiên chúng tôi thấy ID nút đó
							if(mapSupportBeta.get(parent.itemID) == null){
								// chỉ cần thêm số lượng đường dẫn
								mapSupportBeta.put(parent.itemID, pathCount);
							}
                            else{
								// otherwise, make the sum with the value already stored
								mapSupportBeta.put(parent.itemID, mapSupportBeta.get(parent.itemID) + pathCount);
							}
							parent = parent.parent;
						}
						// add the path to the list of prefixpaths
						prefixPaths.add(prefixPath);
					}
					// We will look for the next prefixpath
					path = path.nodeLink;
				}

				
				// ===== FP-CLOSE ======
				// concatenate Beta (Head) with the item "item" (i) to check 
				// for closure
				int[] headWithP = new int[prefixLength+1];
				System.arraycopy(prefix, 0, headWithP, 0, prefixLength+1);
				
				// Sort Head U {item} according to the original header list total order on items
				// sort item in the transaction by descending order of support
				sortOriginalOrder(headWithP, prefixLength+1);

				//======= DEBUG ========
				if(DEBUG) {
					System.out.println(" CHECK2 : " + Arrays.toString(headWithP) + " sup=" + betaSupport);
				}
				//========== END DEBUG =======
				
				// CHECK IF HEAD U P IS A SUBSET OF A CFI ACCORDING TO THE CFI-TREE
				if(cfiTree.passSubsetChecking(headWithP, prefixLength+1, betaSupport)) {
					
					if(DEBUG) {
						System.out.println("    passed!");
					}
					// (B) Construct beta's conditional FP-Tree using its prefix path
					// Create the tree.
					FPTree treeBeta = new FPTree();
					// Add each prefixpath in the FP-tree.
					for(List<FPNode> prefixPath : prefixPaths){
						treeBeta.addPrefixPath(prefixPath, mapSupportBeta, minSupportRelative); 
					}  
					// Mine recursively the Beta tree if the root has child(s)
					if(treeBeta.root.childs.size() > 0){

						// Create the header list.
						treeBeta.createHeaderList(originalMapSupport); 
						
						// recursive call
						fpclose(treeBeta, prefix, prefixLength+1, betaSupport, mapSupportBeta);
					}
					// if the tree is empty we still need to try to save the 
					// itemset
					if(cfiTree.passSubsetChecking(headWithP, prefixLength+1, betaSupport)) {
						saveItemset(headWithP, prefixLength+1, betaSupport);
					}
				}
				else {
					if(DEBUG) {
						System.out.println("     failed!");
					}
//					// OPTIMIZATION ONLY IN FPCLOSE:  IF THE CLOSURE CHECKING iS NOT PASSED
//					// WE STOP THIS LOOP BECAUSE THE NEXT ITEMS WILL NOT PASS IT EITHER
//					break;
				}
			}
		}
	}
